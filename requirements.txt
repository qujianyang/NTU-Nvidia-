# INSTALLATION GUIDE
# ==============================================================================
# This project uses a mix of standard Python packages and GPU-accelerated
# libraries from the NVIDIA RAPIDS ecosystem. Due to complex dependencies,
# it is STRONGLY recommended to use the Conda package manager for setup.
#
# Follow these steps to create a clean and correct environment:
#
# 1. Create a new Conda environment:
#    conda create -n nvidia_stack python=3.10
#
# 2. Activate the environment:
#    conda activate nvidia_stack
#
# 3. Install the core GPU stack (RAPIDS, FAISS, PyTorch) using Conda.
#    This is the most critical step. This command installs versions that are
#    compatible with each other and the required CUDA toolkit.
#
#    conda install -c rapidsai -c conda-forge -c nvidia -c pytorch \
#      cudf cugraph faiss-gpu nx-cugraph \
#      pytorch torchvision torchaudio pytorch-cuda=12.1 --yes
#
# 4. Install the remaining application packages from this file using pip:
#    pip install -r requirements.txt
#
# If you encounter dependency conflicts in the future, it is best to recreate
# the environment by following these steps.
# ==============================================================================
